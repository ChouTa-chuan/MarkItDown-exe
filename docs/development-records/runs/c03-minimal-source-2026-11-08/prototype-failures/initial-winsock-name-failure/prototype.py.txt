from pathlib import Path
import subprocess,json,os
base=Path('/workspace/markitdown-minimal-prototype');base.mkdir(exist_ok=True)
source=Path('/workspace/markitdown-minimal-sources');tools=Path('/workspace/markitdown-native-tools/llvm-mingw-20261104-ucrt-ubuntu-22.04-x86_64');prefix=base/'prefix';cmake='/workspace/markitdown-native-build-venv/bin/cmake';ninja='/workspace/markitdown-native-build-venv/bin/ninja'
chain=base/'toolchain.cmake';chain.write_text(f'''set(CMAKE_SYSTEM_NAME Windows)
set(CMAKE_SYSTEM_PROCESSOR AMD64)
set(CMAKE_C_COMPILER "{tools}/bin/x86_64-w64-mingw32-clang")
set(CMAKE_CXX_COMPILER "{tools}/bin/x86_64-w64-mingw32-clang++")
set(CMAKE_RC_COMPILER "{tools}/bin/x86_64-w64-mingw32-windres")
set(CMAKE_AR "{tools}/bin/llvm-ar")
set(CMAKE_RANLIB "{tools}/bin/llvm-ranlib")
set(CMAKE_STRIP "{tools}/bin/llvm-strip")
set(CMAKE_FIND_ROOT_PATH "{prefix};{tools}/x86_64-w64-mingw32")
set(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM NEVER)
set(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY ONLY)
set(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE ONLY)
set(CMAKE_FIND_ROOT_PATH_MODE_PACKAGE ONLY)
set(CMAKE_EXE_LINKER_FLAGS "-static -Wl,--no-insert-timestamp")
''')
recipes={
'zlib':['-DZLIB_BUILD_SHARED=OFF','-DZLIB_BUILD_STATIC=ON','-DZLIB_BUILD_TESTING=OFF'],
'libpng':['-DZLIB_LIBRARY='+str(prefix/'lib/libzs.a'),'-DZLIB_INCLUDE_DIR='+str(prefix/'include'),'-DPNG_SHARED=OFF','-DPNG_STATIC=ON','-DPNG_TESTS=OFF','-DPNG_TOOLS=OFF','-DPNG_DEBUG=OFF','-DPNG_HARDWARE_OPTIMIZATIONS=OFF'],
'leptonica':['-DZLIB_LIBRARY='+str(prefix/'lib/libzs.a'),'-DZLIB_INCLUDE_DIR='+str(prefix/'include'),'-DBUILD_SHARED_LIBS=OFF','-DBUILD_PROG=OFF','-DSW_BUILD=OFF','-DSTRICT_CONF=ON','-DENABLE_ZLIB=ON','-DENABLE_PNG=ON',*['-DENABLE_'+v+'=OFF' for v in ['GIF','JPEG','TIFF','WEBP','OPENJPEG']]],
 'tesseract':['-DCMAKE_CXX_FLAGS=-include string -include cstdlib','-DLEPT_TIFF_RESULT=1','-DBUILD_SHARED_LIBS=OFF','-DBUILD_TRAINING_TOOLS=OFF','-DBUILD_TESTS=OFF','-DOPENMP_BUILD=OFF','-DGRAPHICS_DISABLED=ON','-DDISABLED_LEGACY_ENGINE=ON','-DENABLE_NATIVE=OFF','-DENABLE_LTO=OFF','-DENABLE_CCACHE=OFF','-DENABLE_PRECOMPILED_HEADERS=OFF','-DINSTALL_CONFIGS=OFF','-DDISABLE_TIFF=ON','-DDISABLE_ARCHIVE=ON','-DDISABLE_CURL=ON','-DUSE_SYSTEM_ICU=OFF','-DCMAKE_DISABLE_FIND_PACKAGE_PkgConfig=TRUE','-DCMAKE_EXE_LINKER_FLAGS_RELEASE=-Wl,-Map,'+str(base/'tesseract-link.map')]}
commands=[]
def call(label,args):
 commands.append(dict(label=label,argv=args));(base/'commands.json').write_text(json.dumps(commands,indent=2)+'\n')
 print(label,flush=True)
 with (base/(label+'.log')).open('wb') as log:subprocess.run(args,stdout=log,stderr=subprocess.STDOUT,check=True,env={**os.environ,'SOURCE_DATE_EPOCH':'1784870400'})
for name,flags in recipes.items():
 src=Path('/workspace/markitdown-native-source-5.5.3') if name=='tesseract' else source/name;bld=base/('build-'+name)
 call(name+'-configure',[cmake,'-S',str(src),'-B',str(bld),'-G','Ninja','-DCMAKE_MAKE_PROGRAM='+ninja,'-DCMAKE_TOOLCHAIN_FILE='+str(chain),'-DCMAKE_BUILD_TYPE=Release','-DCMAKE_INSTALL_PREFIX='+str(prefix),'-DCMAKE_INSTALL_LIBDIR=lib','-DCMAKE_PREFIX_PATH='+str(prefix),*flags])
 call(name+'-build',[cmake,'--build',str(bld),'--parallel','2'])
 call(name+'-install',[cmake,'--install',str(bld)])
print('Actual Windows PE built without execution',base/'build-tesseract/bin/tesseract.exe',flush=True)
